    1. Tracing

    • Need

      securityContext: capabilities: add:

      • SYS_PTRACE

      added to the pod to allow for strace and ptrace to work. Example:

        apiVersion: v1
  kind: Pod
  metadata:
    name: <POD_NAME>
    namespace: default
  spec:
    containers:
      image: <IMAGE>
      imagePullPolicy: IfNotPresent
      name: web
      command: ["/bin/sleep"]
      args: ["1000000"]
      securityContext:
        capabilities:
          add:
          - SYS_PTRACE
      ports:
      - containerPort: 80
        name: http
        protocol: TCP
      resources:
        limits:
          cpu: "40"
          memory: 100G
        requests:
          cpu: "38"
          memory: 100G
      terminationMessagePath: /dev/termination-log
      terminationMessagePolicy: File
      volumeMounts:
      - mountPath: /var/run/secrets/kubernetes.io/serviceaccount
        name: kube-api-access-qtpks
        readOnly: true
